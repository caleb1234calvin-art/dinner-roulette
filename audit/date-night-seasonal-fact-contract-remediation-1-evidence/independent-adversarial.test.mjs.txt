test('schema and manifest failures preserve atomic file bytes',()=>{
  const dir=mkdtempSync(join(tmpdir(),'independent-seasonal-'));try{
    const path=join(dir,'bundle.json'), s=buildSnapshot(one());writeSnapshot(path,s);const bytes=readFileSync(path);
    for(const edit of [x=>x.manifest.schemaVersion='wrong',x=>x.manifest.artifacts.runtime.bytes++]){
      const bad=structuredClone(s);edit(bad);assert.throws(()=>writeSnapshot(path,bad));assert.deepEqual(readFileSync(path),bytes);
    }
  }finally{rmSync(dir,{recursive:true,force:true});}
